# CMDB MVP Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 让 AI Agent 具备"查资产"能力——通过 `query_cmdb` 工具查询服务元数据（Owner、团队、集群、依赖），数据存本地 YAML 文件，只读不写。

**Architecture:** 三层 adapter 模式：`infra/cmdb` 读 YAML 并暴露自有 DTO，`CMDBAdapter` 做 DTO→Info 转换并实现 `ai/cmdb.ServiceRepository` interface，`main.go` 组装注入。AI Agent 通过结构化输入的 `query_cmdb` 工具查询，结果以 JSON 进入 ToolItems 流程。ContextEngine MVP 阶段不直接依赖 CMDB。

**Tech Stack:** Go 1.24, GoFrame v2, gopkg.in/yaml.v3, cloudwego/eino

```go
// internal/infra/cmdb/store.go
package cmdb

import (
	"fmt"
	"os"
	"strings"
	"sync"

	"gopkg.in/yaml.v3"
)

// serviceInfoDTO 内部查询用的扁平结构（infra 自有）
type serviceInfoDTO struct {
	Name         string
	DisplayName  string
	Owner        string
	Team         string
	Cluster      string
	Env          string
	Region       string
	Language     string
	Port         int
	Dependencies []string
	Description  string
	OnCall       string
	LastDeploy   string
	Tags         []string
}

// YAMLLoader 从 YAML 文件加载服务数据，提供只读查询
type YAMLLoader struct {
	path     string
	mu       sync.RWMutex
	services map[string]serviceInfoDTO
}

// NewYAMLLoader 创建并加载 YAML 文件
func NewYAMLLoader(path string) (*YAMLLoader, error) {
	l := &YAMLLoader{
		path:     path,
		services: make(map[string]serviceInfoDTO),
	}
	if err := l.load(); err != nil {
		return nil, err
	}
	return l, nil
}

func (l *YAMLLoader) load() error {
	data, err := os.ReadFile(l.path)
	if err != nil {
		return fmt.Errorf("cmdb: read yaml: %w", err)
	}
	var file cmdbFile
	if err := yaml.Unmarshal(data, &file); err != nil {
		return fmt.Errorf("cmdb: parse yaml: %w", err)
	}
	l.mu.Lock()
	defer l.mu.Unlock()
	l.services = make(map[string]serviceInfoDTO, len(file.Services))
	for _, s := range file.Services {
		dto := serviceInfoDTO{
			Name:         s.Name,
			DisplayName:  s.DisplayName,
			Owner:        s.Owner,
			Team:         s.Team,
			Cluster:      s.Cluster,
			Env:          s.Env,
			Region:       s.Region,
			Language:     s.Language,
			Port:         s.Port,
			Dependencies: s.Dependencies,
			Description:  s.Description,
			OnCall:       s.OnCall,
			LastDeploy:   s.LastDeploy,
			Tags:         s.Tags,
		}
		if dto.Name != "" {
			l.services[dto.Name] = dto
		}
	}
	return nil
}

// Reload 重新加载 YAML 文件（热更新）
func (l *YAMLLoader) Reload() error {
	return l.load()
}

// GetService 查询单个服务
func (l *YAMLLoader) GetService(name string) (serviceInfoDTO, bool) {
	l.mu.RLock()
	defer l.mu.RUnlock()
	svc, ok := l.services[name]
	return svc, ok
}

// ListAll 返回所有服务
func (l *YAMLLoader) ListAll() []serviceInfoDTO {
	l.mu.RLock()
	defer l.mu.RUnlock()
	result := make([]serviceInfoDTO, 0, len(l.services))
	for _, svc := range l.services {
		result = append(result, svc)
	}
	return result
}

// SearchServices 模糊搜索（按 name/display_name/description/tag）
func (l *YAMLLoader) SearchServices(keyword string, limit int) []serviceInfoDTO {
	l.mu.RLock()
	defer l.mu.RUnlock()
	keyword = strings.ToLower(keyword)
	var results []serviceInfoDTO
	for _, svc := range l.services {
		if matchService(svc, keyword) {
			results = append(results, svc)
			if limit > 0 && len(results) >= limit {
				break
			}
		}
	}
	return results
}

func matchService(svc serviceInfoDTO, keyword string) bool {
	if strings.Contains(strings.ToLower(svc.Name), keyword) {
		return true
	}
	if strings.Contains(strings.ToLower(svc.DisplayName), keyword) {
		return true
	}
	if strings.Contains(strings.ToLower(svc.Description), keyword) {
		return true
	}
	for _, tag := range svc.Tags {
		if strings.Contains(strings.ToLower(tag), keyword) {
			return true
		}
	}
	return false
}

// ListServicesByCluster 查询集群下所有服务
func (l *YAMLLoader) ListServicesByCluster(cluster string) []serviceInfoDTO {
	l.mu.RLock()
	defer l.mu.RUnlock()
	var results []serviceInfoDTO
	for _, svc := range l.services {
		if svc.Cluster == cluster {
			results = append(results, svc)
		}
	}
	return results
}

// ListServicesByTeam 查询团队下所有服务
func (l *YAMLLoader) ListServicesByTeam(team string) []serviceInfoDTO {
	l.mu.RLock()
	defer l.mu.RUnlock()
	var results []serviceInfoDTO
	for _, svc := range l.services {
		if svc.Team == team {
			results = append(results, svc)
		}
	}
	return results
}

// GetDependents 反向计算：谁依赖该服务
func (l *YAMLLoader) GetDependents(name string) []string {
	l.mu.RLock()
	defer l.mu.RUnlock()
	var dependents []string
	for _, svc := range l.services {
		for _, dep := range svc.Dependencies {
			if dep == name {
				dependents = append(dependents, svc.Name)
				break
			}
		}
	}
	return dependents
}
```

```go
// internal/infra/cmdb/adapter.go
package cmdb

import (
	"SuperBizAgent/internal/ai/cmdb"
)

// CMDBAdapter 实现 ai/cmdb.ServiceRepository interface
// 内部做 DTO → Info 转换，不泄露 infra 细节
type CMDBAdapter struct {
	loader *YAMLLoader
}

// NewCMDBAdapter 创建 adapter
func NewCMDBAdapter(loader *YAMLLoader) *CMDBAdapter {
	return &CMDBAdapter{loader: loader}
}

func (a *CMDBAdapter) GetService(name string) (*cmdb.ServiceInfo, bool) {
	dto, ok := a.loader.GetService(name)
	if !ok {
		return nil, false
	}
	info := dtoToInfo(dto, a.loader)
	return &info, true
}

func (a *CMDBAdapter) SearchServices(keyword string, limit int) []cmdb.ServiceInfo {
	dtos := a.loader.SearchServices(keyword, limit)
	return dtosToInfos(dtos, a.loader)
}

func (a *CMDBAdapter) ListServicesByCluster(cluster string) []cmdb.ServiceInfo {
	dtos := a.loader.ListServicesByCluster(cluster)
	return dtosToInfos(dtos, a.loader)
}

func (a *CMDBAdapter) ListServicesByTeam(team string) []cmdb.ServiceInfo {
	dtos := a.loader.ListServicesByTeam(team)
	return dtosToInfos(dtos, a.loader)
}

func (a *CMDBAdapter) GetDependents(name string) []string {
	return a.loader.GetDependents(name)
}

func (a *CMDBAdapter) ListAll() []cmdb.ServiceInfo {
	dtos := a.loader.ListAll()
	return dtosToInfos(dtos, a.loader)
}

func dtoToInfo(dto serviceInfoDTO, loader *YAMLLoader) cmdb.ServiceInfo {
	return cmdb.ServiceInfo{
		Name:         dto.Name,
		DisplayName:  dto.DisplayName,
		Owner:        dto.Owner,
		Team:         dto.Team,
		Cluster:      dto.Cluster,
		Env:          dto.Env,
		Region:       dto.Region,
		Language:     dto.Language,
		Port:         dto.Port,
		Dependencies: dto.Dependencies,
		Dependents:   loader.GetDependents(dto.Name),
		Description:  dto.Description,
		OnCall:       dto.OnCall,
		LastDeploy:   dto.LastDeploy,
		Tags:         dto.Tags,
	}
}

func dtosToInfos(dtos []serviceInfoDTO, loader *YAMLLoader) []cmdb.ServiceInfo {
	result := make([]cmdb.ServiceInfo, 0, len(dtos))
	for _, dto := range dtos {
		result = append(result, dtoToInfo(dto, loader))
	}
	return result
}
```

```go
// internal/ai/cmdb/service.go
package cmdb

import (
	"fmt"
	"strings"
)

// FormatServiceInfo 将单个服务格式化为可读字符串
func FormatServiceInfo(svc *ServiceInfo) string {
	var sb strings.Builder
	sb.WriteString(fmt.Sprintf("服务: %s", svc.Name))
	if svc.DisplayName != "" {
		sb.WriteString(fmt.Sprintf(" (%s)", svc.DisplayName))
	}
	sb.WriteString("\n")
	sb.WriteString(fmt.Sprintf("  负责人: %s\n", svc.Owner))
	sb.WriteString(fmt.Sprintf("  团队: %s\n", svc.Team))
	sb.WriteString(fmt.Sprintf("  集群: %s\n", svc.Cluster))
	sb.WriteString(fmt.Sprintf("  环境: %s\n", svc.Env))
	if svc.Region != "" {
		sb.WriteString(fmt.Sprintf("  地域: %s\n", svc.Region))
	}
	if svc.Language != "" {
		sb.WriteString(fmt.Sprintf("  技术栈: %s\n", svc.Language))
	}
	if len(svc.Dependencies) > 0 {
		sb.WriteString(fmt.Sprintf("  上游依赖: %s\n", strings.Join(svc.Dependencies, ", ")))
	}
	if len(svc.Dependents) > 0 {
		sb.WriteString(fmt.Sprintf("  下游被依赖: %s\n", strings.Join(svc.Dependents, ", ")))
	}
	if svc.Description != "" {
		sb.WriteString(fmt.Sprintf("  描述: %s\n", svc.Description))
	}
	if svc.OnCall != "" {
		sb.WriteString(fmt.Sprintf("  On-Call: %s\n", svc.OnCall))
	}
	if svc.LastDeploy != "" {
		sb.WriteString(fmt.Sprintf("  最近部署: %s\n", svc.LastDeploy))
	}
	if len(svc.Tags) > 0 {
		sb.WriteString(fmt.Sprintf("  标签: %s\n", strings.Join(svc.Tags, ", ")))
	}
	return sb.String()
}

// FormatServiceList 将服务列表格式化为可读字符串
func FormatServiceList(services []ServiceInfo) string {
	if len(services) == 0 {
		return "未找到匹配的服务"
	}
	var sb strings.Builder
	sb.WriteString(fmt.Sprintf("共 %d 个服务:\n", len(services)))
	for i, svc := range services {
		sb.WriteString(fmt.Sprintf("%d. %s", i+1, svc.Name))
		if svc.DisplayName != "" {
			sb.WriteString(fmt.Sprintf(" (%s)", svc.DisplayName))
		}
		sb.WriteString(fmt.Sprintf(" — %s / %s", svc.Owner, svc.Team))
		if len(svc.Tags) > 0 {
```

```go
// internal/ai/tools/query_cmdb.go
package tools

import (
	"SuperBizAgent/internal/ai/cmdb"
	"context"
	"encoding/json"
	"fmt"

	"github.com/cloudwego/eino/components/tool"
	"github.com/cloudwego/eino/components/tool/utils"
)

type QueryCMDBInput struct {
	Action      string `json:"action" jsonschema:"description=查询动作：get_service(查单个服务), search(模糊搜索), list_by_cluster(集群查询), list_by_team(团队查询), get_dependencies(查依赖), list_all(列全部)"`
	ServiceName string `json:"service_name,omitempty" jsonschema:"description=服务名称，用于 get_service / get_dependencies"`
	Cluster     string `json:"cluster,omitempty" jsonschema:"description=集群名称，用于 list_by_cluster"`
	Team        string `json:"team,omitempty" jsonschema:"description=团队名称，用于 list_by_team"`
	Keyword     string `json:"keyword,omitempty" jsonschema:"description=搜索关键词，用于 search"`
	Limit       int    `json:"limit,omitempty" jsonschema:"description=返回结果数量上限，默认 10"`
}

type QueryCMDBOutput struct {
	Success  bool              `json:"success"`
	Degraded bool              `json:"degraded,omitempty"`
	Action   string            `json:"action"`
	Services []cmdb.ServiceInfo `json:"services,omitempty"`
	Service  *cmdb.ServiceInfo  `json:"service,omitempty"`
	Upstream []string          `json:"upstream,omitempty"`
	Downstream []string        `json:"downstream,omitempty"`
	Message  string            `json:"message,omitempty"`
	Error    string            `json:"error,omitempty"`
}

func NewQueryCMDBTool(repo cmdb.ServiceRepository) tool.InvokableTool {
	t, err := utils.InferOptionableTool(
		"query_cmdb",
		"查询服务资产信息（CMDB）。可查询服务详情、负责人、团队、集群、依赖关系等。用于回答'谁负责这个服务'、'服务部署在哪'、'依赖谁'等资产相关问题。",
		func(ctx context.Context, input *QueryCMDBInput, opts ...tool.Option) (output string, err error) {
			if repo == nil {
				return marshalOutput(QueryCMDBOutput{
					Success:  false,
					Degraded: true,
					Action:   input.Action,
					Error:    "CMDB not available",
					Message:  "CMDB 未启用或未配置",
				})
			}

			limit := input.Limit
			if limit <= 0 {
				limit = 10
			}

			switch input.Action {
			case "get_service":
				svc, ok := repo.GetService(input.ServiceName)
				if !ok {
					return marshalOutput(QueryCMDBOutput{
						Success: true,
						Action:  input.Action,
						Message: fmt.Sprintf("未找到服务 %q", input.ServiceName),
					})
				}
				return marshalOutput(QueryCMDBOutput{
					Success: true,
					Action:  input.Action,
					Service: svc,
				})

			case "search":
				results := repo.SearchServices(input.Keyword, limit)
				return marshalOutput(QueryCMDBOutput{
					Success:  true,
					Action:   input.Action,
					Services: results,
					Message:  fmt.Sprintf("找到 %d 个服务", len(results)),
				})

			case "list_by_cluster":
				results := repo.ListServicesByCluster(input.Cluster)
				return marshalOutput(QueryCMDBOutput{
					Success:  true,
					Action:   input.Action,
					Services: results,
					Message:  fmt.Sprintf("集群 %s 下有 %d 个服务", input.Cluster, len(results)),
				})

			case "list_by_team":
				results := repo.ListServicesByTeam(input.Team)
				return marshalOutput(QueryCMDBOutput{
					Success:  true,
					Action:   input.Action,
					Services: results,
					Message:  fmt.Sprintf("团队 %s 下有 %d 个服务", input.Team, len(results)),
				})

			case "get_dependencies":
				svc, ok := repo.GetService(input.ServiceName)
				if !ok {
					return marshalOutput(QueryCMDBOutput{
						Success: true,
						Action:  input.Action,
						Message: fmt.Sprintf("未找到服务 %q", input.ServiceName),
					})
				}
				downstream := repo.GetDependents(input.ServiceName)
				return marshalOutput(QueryCMDBOutput{
					Success:    true,
					Action:     input.Action,
					Service:    svc,
					Upstream:   svc.Dependencies,
					Downstream: downstream,
				})

			case "list_all":
				results := repo.ListAll()
				return marshalOutput(QueryCMDBOutput{
					Success:  true,
					Action:   input.Action,
					Services: results,
					Message:  fmt.Sprintf("共 %d 个服务", len(results)),
				})

			default:
				return marshalOutput(QueryCMDBOutput{
					Success: false,
					Action:  input.Action,
					Error:   fmt.Sprintf("unknown action: %s", input.Action),
					Message: fmt.Sprintf("不支持的查询动作: %s", input.Action),
				})
			}
		})
	if err != nil {
		return nil
	}
	return t
}

func marshalOutput(out QueryCMDBOutput) (string, error) {
	b, err := json.Marshal(out)
	if err != nil {
		return fmt.Sprintf(`{"success":false,"error":"marshal failed: %s"}`, err.Error()), nil
	}
	return string(b), nil
}
```

```go
		return map[string]interface{}{"success": false, "message": "CMDB 未启用"}

	if a.repo == nil {
		return map[string]interface{}{"success": false, "message": "CMDB 未启用"}
	}
	svc, ok := a.repo.GetService(name)
	if !ok {
		return map[string]interface{}{"success": true, "message": "未找到服务 " + name}
	}
	return map[string]interface{}{"success": true, "service": svc}
}
```
